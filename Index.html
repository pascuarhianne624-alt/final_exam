Puzzle Game Source Code

Below is the source code of my Puzzle Game project. It is displayed as code only and will not run inside this portfolio. 


let tiles = [1,2,3,4,5,6,7,8,0];

const board = document.getElementById('board');
const msg = document.getElementById('msg');

function draw() {
    board.innerHTML = '';

    tiles.forEach((n, i) => {
        let div = document.createElement('div');

        div.className = n === 0 ? 'tile empty' : 'tile';
        div.textContent = n === 0 ? '' : n;
        div.onclick = () => move(i);

        board.appendChild(div);
    });

    if (tiles.join(',') === '1,2,3,4,5,6,7,8,0') {
        msg.textContent = 'Puzzle Solved!';
    }
}

function move(i) {
    let e = tiles.indexOf(0);

    let can = [e - 3, e + 3];

    if (e % 3 !== 0) can.push(e - 1);
    if (e % 3 !== 2) can.push(e + 1);

    if (can.includes(i)) {
        [tiles[e], tiles[i]] = [tiles[i], tiles[e]];

        msg.textContent = 'Fix it 1-8!';
        draw();
    }
}

function shuffle() {
    for (let i = 0; i < 200; i++) {
        let e = tiles.indexOf(0);

        let moves = [e - 3, e + 3];

        if (e % 3 !== 0) moves.push(e - 1);
        if (e % 3 !== 2) moves.push(e + 1);

        moves = moves.filter(m => m >= 0 && m < 9);

        let r = moves[Math.floor(Math.random() * moves.length)];

        [tiles[e], tiles[r]] = [tiles[r], tiles[e]];
    }

    msg.textContent = 'Fix it 1-8!';
    draw();
}

draw();
